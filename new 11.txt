import hashlib
import json

def hash_result(result):
    normalized = json.dumps(result, sort_keys=True)
    return hashlib.md5(normalized.encode()).hexdigest()
injection_checker.py
def detect_sql_injection(sql):
    forbidden = ["DROP", "DELETE", "INSERT", ";--"]
    for word in forbidden:
        assert word not in sql.upper()
